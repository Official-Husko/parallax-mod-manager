to get more details use public steam endpoints for mods:

curl -X POST \
  -d "itemcount=1" \
  -d "publishedfileids[0]=1780481482" \
  https://api.steampowered.com/ISteamRemoteStorage/GetPublishedFileDetails/v1/

curl -X POST \
  -d "itemcount=3" \
  -d "publishedfileids[0]=123" \
  -d "publishedfileids[1]=456" \
  -d "publishedfileids[2]=789" \
  https://api.steampowered.com/ISteamRemoteStorage/GetPublishedFileDetails/v1/
